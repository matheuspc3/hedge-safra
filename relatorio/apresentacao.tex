\begin{frame}{Modelagem — Ornstein-Uhlenbeck + CDD $\to$ Produtividade}

  \textbf{Processo de temperatura:}
  $\displaystyle dT(t) = \theta (\mu - T(t)) dt + \sigma dW(t)$
  \hfill Meia-vida: \destaque{3,2--3,6 dias}

  \vspace{0.15cm}
  \textbf{CDD (Cooling Degree Days):}
  $\displaystyle \text{CDD} = \sum_{\text{DEZ-FEV}} \max(0, T_{\text{média}} - 25^\circ\text{C})$

  \vspace{0.15cm}
  \textbf{Perda de produtividade (piecewise):}
  $\displaystyle Y = Y_{\max} \times \bigl[1 - \min\bigl(\gamma \cdot \max(0, \text{CDD} - K_{\text{prod}}),\, \text{perda}_{\max}\bigr)\bigr]$

  \vspace{0.2cm}
